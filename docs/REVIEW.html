open Finset MvPolynomial

/-- The chromatic sum of an edge list with positive integer vertex weights.
Loops are allowed in the data and force the sum to vanish. -/
def weightedCSF (N : ℕ) {n : ℕ} (w : Fin n → ℕ) (E : Finset (Fin n × Fin n)) : Pol N :=
  ∑ c : Fin n → Fin N, if ∀ e ∈ E, c e.1 ≠ c e.2 then ∏ v, X (c v) ^ w v else 0

/-- Identify the last vertex with `u`. -/
def contractVertex {n : ℕ} (u : Fin n) : Fin (n + 1) → Fin n := Fin.lastCases u id

/-- Add the last vertex's weight to `u`. -/
def contractWeight {n : ℕ} (w : Fin (n + 1) → ℕ) (u : Fin n) : Fin n → ℕ :=
  fun v => w v.castSucc + if v = u then w (Fin.last n) else 0

/-- Images of the remaining edges, including loops when appropriate. -/
def contractEdges {n : ℕ} (E : Finset (Fin (n + 1) × Fin (n + 1))) (u : Fin n) :
    Finset (Fin n × Fin n) := E.image fun e => (contractVertex u e.1, contractVertex u e.2)

lemma weightedCSF_loop (N : ℕ) {n : ℕ} (w : Fin n → ℕ)
    (E : Finset (Fin n × Fin n)) (v : Fin n) (hv : (v, v) ∈ E) : weightedCSF N w E = 0 := by
  classical
  apply sum_eq_zero
  intro c _
  rw [ite_eq_right]
  intro hc
  exact hc (v, v) hv rfl

lemma weightedCSF_empty (N : ℕ) : weightedCSF N (fun v : Fin 0 => v.elim0) ∅ = 1 := by
  classical
  simp [weightedCSF]

lemma weightedCSF_congr (N : ℕ) {n : ℕ} {w w' : Fin n → ℕ}
    {E E' : Finset (Fin n × Fin n)} (hw : w = w')
    (hE : ∀ c : Fin n → Fin N, (∀ e ∈ E, c e.1 ≠ c e.2) ↔ (∀ e ∈ E', c e.1 ≠ c e.2)) :
    weightedCSF N w E = weightedCSF N w' E' := by
  classical
  subst w'
  unfold weightedCSF
  simp_rw [hE]

lemma weighted_product_contract {N n : ℕ} (w : Fin (n + 1) → ℕ) (u : Fin n)
    (c : Fin n → Fin N) :
    (∏ v : Fin (n + 1), (X (Fin.snoc (α := fun _ => Fin N) c (c u) v) : Pol N) ^ w v) =
      ∏ v : Fin n, X (c v) ^ contractWeight w u v := by
  classical
  rw [Fin.prod_univ_castSucc]
  simp only [Fin.snoc_castSucc, Fin.snoc_last, contractWeight, pow_add, prod_mul_distrib]
  congr 1
  simp

lemma weightedCSF_equal_last (N : ℕ) {n : ℕ} (w : Fin (n + 1) → ℕ)
    (E : Finset (Fin (n + 1) × Fin (n + 1))) (u : Fin n) :
    (∑ c : Fin (n + 1) → Fin N,
      if (∀ e ∈ E, c e.1 ≠ c e.2) ∧ c (Fin.last n) = c u.castSucc then
        ∏ v, (X (c v) : Pol N) ^ w v else 0) =
      weightedCSF N (contractWeight w u) (contractEdges E u) := by
  classical
  have hsplit : ∀ F : (Fin (n + 1) → Fin N) → Pol N,
      ∑ c, F c = ∑ y : Fin N, ∑ c : Fin n → Fin N, F (Fin.snoc (α := fun _ => Fin N) c y) := by
    intro F
    rw [← Fintype.sum_prod_type']
    exact Fintype.sum_equiv (Fin.snocEquiv (n := n) (fun _ => Fin N)).symm _ _ (by intro c; simp [Fin.snocEquiv])
  rw [hsplit, sum_comm]
  unfold weightedCSF
  apply sum_congr rfl
  intro c _
  rw [sum_eq_single (c u)]
  · have hproper : (∀ e ∈ E, Fin.snoc (α := fun _ => Fin N) c (c u) e.1 ≠ Fin.snoc (α := fun _ => Fin N) c (c u) e.2) ↔
        ∀ e ∈ contractEdges E u, c e.1 ≠ c e.2 := by
      have hc : ∀ v, Fin.snoc (α := fun _ => Fin N) c (c u) v = c (contractVertex u v) := by
        intro v; induction v using Fin.lastCases <;> simp [contractVertex]
      simp only [contractEdges, forall_mem_image, hc]
    simp only [Fin.snoc_last, Fin.snoc_castSucc, and_true, hproper]
    rw [weighted_product_contract]
  · intro y _ hy
    simp [hy]
  · simp

/-- Deletion-contraction on an edge incident to the last vertex. -/
theorem weightedCSF_delete_contract (N : ℕ) {n : ℕ} (w : Fin (n + 1) → ℕ)
    (E : Finset (Fin (n + 1) × Fin (n + 1))) (u : Fin n) :
    weightedCSF N w (insert (u.castSucc, Fin.last n) E) = weightedCSF N w E -
      weightedCSF N (contractWeight w u) (contractEdges E u) := by
  classical
  rw [← weightedCSF_equal_last]
  unfold weightedCSF
  rw [← sum_sub_distrib]
  apply sum_congr rfl
  intro c _
  simp only [forall_mem_insert]
  split_ifs <;> simp_all [Ne, eq_comm]

/-- An isolated last vertex contributes its power sum. -/
theorem weightedCSF_isolated (N : ℕ) {n : ℕ} (w : Fin (n + 1) → ℕ)
    (E : Finset (Fin n × Fin n)) :
    weightedCSF N w (E.image fun e => (e.1.castSucc, e.2.castSucc)) =
      (∑ y : Fin N, (X y : Pol N) ^ w (Fin.last n)) * weightedCSF N (fun v => w v.castSucc) E := by
  classical
  unfold weightedCSF
  rw [mul_sum]
  have hsplit : ∀ F : (Fin (n + 1) → Fin N) → Pol N,
      ∑ c, F c = ∑ y : Fin N, ∑ c : Fin n → Fin N, F (Fin.snoc (α := fun _ => Fin N) c y) := by
    intro F
    rw [← Fintype.sum_prod_type']
    exact Fintype.sum_equiv (Fin.snocEquiv (n := n) (fun _ => Fin N)).symm _ _ (by intro c; simp [Fin.snocEquiv])
  rw [hsplit, sum_comm]
  apply sum_congr rfl
  intro c _
  have hproper : ∀ y : Fin N,
      (∀ e ∈ E.image (fun e => (e.1.castSucc, e.2.castSucc)),
        Fin.snoc (α := fun _ => Fin N) c y e.1 ≠ Fin.snoc (α := fun _ => Fin N) c y e.2) ↔
        (∀ e ∈ E, c e.1 ≠ c e.2) := by
    intro y
    simp only [forall_mem_image, Fin.snoc_castSucc]
  simp_rw [hproper]
  split_ifs
  · simp only [Fin.prod_univ_castSucc, Fin.snoc_castSucc, Fin.snoc_last]
    rw [← mul_sum]
    ring
  · simp

/-- Relate an edge certificate to a simple graph's CSF. -/
theorem chromSym_eq_weightedCSF (N : ℕ) {n : ℕ} (G : SimpleGraph (Fin n))
    (E : Finset (Fin n × Fin n))
    (hE : ∀ u v, G.Adj u v ↔ (u, v) ∈ E ∨ (v, u) ∈ E) :
    chromSym N G = weightedCSF N (fun _ => 1) E := by
  classical
  unfold chromSym weightedCSF
  apply sum_congr rfl
  intro c _
  have hp : IsProper G c ↔ ∀ e ∈ E, c e.1 ≠ c e.2 := by
    constructor
    · intro h e he; exact h _ _ ((hE _ _).mpr (Or.inl he))
    · intro h u v huv
      rcases (hE u v).mp huv with he | he
      · exact h (u, v) he
      · exact (h (v, u) he).symm
  simp [hp]

def edgeGraph {n : ℕ} (E : Finset (Fin n × Fin n)) (hE : ∀ v, (v, v) ∉ E) :

  loopless := ⟨fun v h => h.elim (hE v) (hE v)⟩

# Products of marked graphs and Lemma `tableau-mult`

Words represent arbitrary marked NUIGs by `word_of_nuig`. Composition executes the
right-hand graph first and identifies its outputs with the left-hand graph's inputs.
The two inclusions below describe the graph gluing, including overlapping input and
output cliques and the empty word.
-/

noncomputable section

set_option backward.isDefEq.respectTransparency false

namespace Word

variable {a b c N : ℕ}

/-- Execute `h`, then `g`: the product `G + H` in the paper's matrix order. -/
def comp (h : Word a b) : {c : ℕ} → Word b c → Word a c
  | _, nil => h
  | _, M g => M (comp h g)
  | _, D g => D (comp h g)

lemma tabMat_comp (h : Word a b) : ∀ {c : ℕ} (g : Word b c),
    tabMat N (comp h g) = tabMat N g * tabMat N h
  | _, nil => by simp [comp, tabMat]
  | _, M g => by rw [comp, tabMat, tabMat, tabMat_comp h g, Matrix.mul_assoc]
  | _, D g => by rw [comp, tabMat, tabMat, tabMat_comp h g, Matrix.mul_assoc]

/-- **Lemma `tableau-mult`** for arbitrary marked NUIGs, represented by words. -/
theorem hikMat_comp (h : Word a b) (g : Word b c) :
    hikMat N (graph (comp h g)) (inp (comp h g)) (mark (comp h g)) =
      hikMat N (graph g) (inp g) (mark g) * hikMat N (graph h) (inp h) (mark h) := by
  simp only [hikMat_eq_tabMat, tabMat_comp]

/-- Inclusion of the vertices of the graph executed first. -/
def compLeft (h : Word a b) : {c : ℕ} → (g : Word b c) → Fin (nv h) → Fin (nv (comp h g))
  | _, nil => id
  | _, M g => fun v => (compLeft h g v).castSucc
  | _, D g => compLeft h g

/-- Inclusion of the vertices of the graph executed second; its initial clique is identified
with the outputs of the first graph. -/
def compRight (h : Word a b) : {c : ℕ} → (g : Word b c) → Fin (nv g) → Fin (nv (comp h g))
  | _, nil => mark h
  | _, M g => Fin.snoc (α := fun _ => Fin (nv (comp h g) + 1))
      (fun v => (compRight h g v).castSucc) (Fin.last _)
  | _, D g => compRight h g

@[simp] lemma compRight_cast (h : Word a b) (g : Word b c) (v : Fin (nv g)) :
    compRight h (M g) v.castSucc = (compRight h g v).castSucc := by simp [compRight]

@[simp] lemma compRight_last (h : Word a b) (g : Word b c) :
    compRight h (M g) (Fin.last _) = Fin.last _ := by simp [compRight]

lemma compLeft_injective (h : Word a b) : ∀ {c : ℕ} (g : Word b c),
    Function.Injective (compLeft h g)
  | _, nil => Function.injective_id
  | _, M g => (Fin.castSucc_injective _).comp (compLeft_injective h g)
  | _, D g => compLeft_injective h g

lemma compRight_injective (h : Word a b) : ∀ {c : ℕ} (g : Word b c),
    Function.Injective (compRight h g)
  | _, nil => mark_injective h
  | _, M g => by
    rw [compRight, Fin.snoc_injective_iff]
    exact ⟨(Fin.castSucc_injective _).comp (compRight_injective h g), by
      rintro ⟨v, hv⟩; exact Fin.castSucc_ne_last _ hv⟩
  | _, D g => compRight_injective h g

    mark (comp h g) r = compRight h g (mark g r)
  | _, nil, r => rfl
  | _, M g, r => by
    induction r using Fin.lastCases <;> simp [comp, mark, comp_mark]
  | _, D g, r => comp_mark h g r.succ

@[simp] lemma comp_inp (h : Word a b) : ∀ {c : ℕ} (g : Word b c) (r : Fin a),
    inp (comp h g) r = compLeft h g (inp h r)
  | _, nil, r => rfl
  | _, M g, r => by simp [comp, inp, compLeft, comp_inp]
  | _, D g, r => comp_inp h g r

/-- Exactly the prescribed clique vertices are identified. -/
lemma comp_overlap (h : Word a b) : ∀ {c : ℕ} (g : Word b c) (u : Fin (nv h)) (v : Fin (nv g)),
    compLeft h g u = compRight h g v ↔ ∃ r : Fin b, u = mark h r ∧ v = inp g r
  | _, nil, u, v => by
    change u = mark h v ↔ ∃ r, u = mark h r ∧ v = r
    exact ⟨fun e => ⟨v, e, rfl⟩, by rintro ⟨r, e, rfl⟩; exact e⟩
  | _, M g, u, v => by
    induction v using Fin.lastCases with
    | last =>
      simp only [compLeft, compRight_last, inp, Fin.castSucc_ne_last, false_iff]
      rintro ⟨r, _, hr⟩
      exact Fin.castSucc_ne_last _ hr.symm
    | cast v => simpa [compLeft, inp, compRight] using comp_overlap h g u v
  | _, D g, u, v => comp_overlap h g u v

/-- Every vertex of the composite comes from one of the two graphs. -/
lemma comp_cover (h : Word a b) : ∀ {c : ℕ} (g : Word b c) (v : Fin (nv (comp h g))),
    (∃ u, compLeft h g u = v) ∨ ∃ u, compRight h g u = v
  | _, nil, v => Or.inl ⟨v, rfl⟩
  | _, M g, v => by
    induction v using Fin.lastCases with
    | last => exact Or.inr ⟨Fin.last _, by simp⟩
    | cast v =>
      rcases comp_cover h g v with ⟨u, rfl⟩ | ⟨u, rfl⟩
      · exact Or.inl ⟨u, rfl⟩
      · exact Or.inr ⟨u.castSucc, by simp⟩
  | _, D g, v => comp_cover h g v

/-- The edges of the composite are exactly the images of the edges of the two factors.
Together with `comp_overlap` and `comp_cover`, this identifies `comp` with graph gluing. -/
theorem adj_comp (h : Word a b) : ∀ {c : ℕ} (g : Word b c) (x y : Fin (nv (comp h g))),
    adj (comp h g) x y ↔
      (∃ u v, adj h u v ∧ compLeft h g u = x ∧ compLeft h g v = y) ∨
      (∃ u v, adj g u v ∧ compRight h g u = x ∧ compRight h g v = y)
  | _, nil, x, y => by
    simp only [comp, compLeft, compRight, id_eq, adj]
    constructor
    · intro hxy; exact Or.inl ⟨x, y, hxy, rfl, rfl⟩
    · rintro (⟨u, v, huv, rfl, rfl⟩ | ⟨u, v, huv, rfl, rfl⟩)
      · exact huv
      · exact mark_clique h u v huv
  | _, M g, x, y => by
    induction x using Fin.lastCases with
    | last =>
      induction y using Fin.lastCases with
      | last =>
        simp only [comp, adj, extendAdj_ll]
        constructor
        · exact False.elim
        · rintro (⟨u, v, _, hu, _⟩ | ⟨u, v, huv, hu, hv⟩)
          · exact Fin.castSucc_ne_last _ hu
          · have hu' : u = Fin.last _ := compRight_injective h (M g) (by simpa only [compRight_last] using hu)
            have hv' : v = Fin.last _ := compRight_injective h (M g) (by simpa only [compRight_last] using hv)
            subst u v; exact adj_irrefl (M g) _ huv
      | cast y =>
        simp only [comp, adj, extendAdj_lc]
        constructor
        · rintro ⟨r, hr⟩
          exact Or.inr ⟨Fin.last _, (mark g r).castSucc, by simp, by simp,
            by rw [compRight_cast, ← comp_mark, hr]⟩
        · rintro (⟨u, v, _, hu, _⟩ | ⟨u, v, huv, hu, hv⟩)
          · exact False.elim (Fin.castSucc_ne_last _ hu)
          · have hu' : u = Fin.last _ := compRight_injective h (M g) (by simpa only [compRight_last] using hu)
            subst u
            induction v using Fin.lastCases with
            | last => exact False.elim (adj_irrefl (M g) _ huv)
            | cast v =>
              obtain ⟨r, rfl⟩ := (show ∃ r, mark g r = v from by simpa [adj] using huv)
              exact ⟨r, by rw [comp_mark]; exact Fin.castSucc_injective _ (by simpa only [compRight_cast] using hv)⟩
    | cast x =>
      induction y using Fin.lastCases with
      | last =>
        simp only [comp, adj, extendAdj_cl]
        constructor
        · rintro ⟨r, hr⟩
          exact Or.inr ⟨(mark g r).castSucc, Fin.last _, by simp,
            by rw [compRight_cast, ← comp_mark, hr], by simp⟩
        · rintro (⟨u, v, _, _, hv⟩ | ⟨u, v, huv, hu, hv⟩)
          · exact False.elim (Fin.castSucc_ne_last _ hv)
          · have hv' : v = Fin.last _ := compRight_injective h (M g) (by simpa only [compRight_last] using hv)
            subst v
            induction u using Fin.lastCases with
            | last => exact False.elim (adj_irrefl (M g) _ huv)
            | cast u =>
              obtain ⟨r, rfl⟩ := (show ∃ r, mark g r = u from by simpa [adj] using huv)
              exact ⟨r, by rw [comp_mark]; exact Fin.castSucc_injective _ (by simpa only [compRight_cast] using hu)⟩
      | cast y =>
        simp only [comp, adj, extendAdj_cc]
        rw [adj_comp h g]
        constructor
        · rintro (⟨u, v, huv, hu, hv⟩ | ⟨u, v, huv, hu, hv⟩)
          · exact Or.inl ⟨u, v, huv, congrArg Fin.castSucc hu, congrArg Fin.castSucc hv⟩
          · exact Or.inr ⟨u.castSucc, v.castSucc, by simpa [adj] using huv,
              by rw [compRight_cast, hu], by rw [compRight_cast, hv]⟩
        · rintro (⟨u, v, huv, hu, hv⟩ | ⟨u, v, huv, hu, hv⟩)
          · exact Or.inl ⟨u, v, huv, Fin.castSucc_injective _ hu, Fin.castSucc_injective _ hv⟩
          · induction u using Fin.lastCases with
            | last => exact False.elim (Fin.castSucc_ne_last _ (by simpa only [compRight_last] using hu.symm))
            | cast u =>
              induction v using Fin.lastCases with
              | last => exact False.elim (Fin.castSucc_ne_last _ (by simpa only [compRight_last] using hv.symm))
              | cast v => exact Or.inr ⟨u, v, by simpa [adj] using huv,
                  Fin.castSucc_injective _ (by simpa only [compRight_cast] using hu),
                  Fin.castSucc_injective _ (by simpa only [compRight_cast] using hv)⟩
  | _, D g, x, y => adj_comp h g x y

/-- Computable adjacency for word graphs, used by exact finite example checks. -/
@[instance_reducible] def decidableAdj {a : ℕ} : ∀ {b : ℕ} (w : Word a b), DecidableRel (adj w)
  | _, nil => by dsimp [adj]; infer_instance
  | _, D w => decidableAdj w
  | _, M w => by
    intro u v
    induction u using Fin.lastCases with
    | last =>
      induction v using Fin.lastCases with
      | last => exact isFalse (by simp [adj])
      | cast v =>
          exact decidable_of_iff (∃ r, mark w r = v) (by simp only [adj, extendAdj_lc])
    | cast u =>
      induction v using Fin.lastCases with
      | last =>
          exact decidable_of_iff (∃ r, mark w r = u) (by simp only [adj, extendAdj_cl])
      | cast v =>
        letI := decidableAdj w u v
        exact decidable_of_iff (adj w u v) (by simp only [adj, extendAdj_cc])

instance {a b : ℕ} (w : Word a b) : DecidableRel (graph w).Adj := decidableAdj w

/-- Word graphs are NUIGs, including words that forget all their marked vertices. -/
lemma graph_isNUIG : ∀ {a b : ℕ} (w : Word a b), IsNUIG (graph w)
  | _, _, nil => by
    intro i j hij _ l hil hlj
    exact ⟨ne_of_lt hil, ne_of_lt hlj⟩
  | _, _, D w => graph_isNUIG w
  | _, _, M (b := b) w => by
    intro i j hij hadj l hil hlj
    induction j using Fin.lastCases with
    | cast j =>
      induction i using Fin.lastCases with
      | last => exact False.elim ((not_lt_of_ge (Fin.castSucc_lt_last j).le) hij)
      | cast i =>
        induction l using Fin.lastCases with
        | last => exact False.elim ((not_lt_of_ge (Fin.castSucc_lt_last j).le) hlj)
        | cast l =>
            simpa only [graph, adj, extendAdj_cc] using
              graph_isNUIG w i j (Fin.castSucc_lt_castSucc_iff.mp hij)
                (by simpa only [graph, adj, extendAdj_cc] using hadj) l
                (Fin.castSucc_lt_castSucc_iff.mp hil) (Fin.castSucc_lt_castSucc_iff.mp hlj)
    | last =>
      induction i using Fin.lastCases with
      | last => exact False.elim (lt_irrefl _ hij)
      | cast i =>
        induction l using Fin.lastCases with
        | last => exact False.elim (lt_irrefl _ hlj)
        | cast l =>
          obtain ⟨r, hr⟩ := (show ∃ r, mark w r = i from by simpa only [graph, adj, extendAdj_cl] using hadj)
          have hstart : nv w - b ≤ (l : ℕ) := by
            have hv := mark_val w r
            have hil' : (i : ℕ) < l := Fin.castSucc_lt_castSucc_iff.mp hil
            rw [hr] at hv
            omega
          have hb := level_le_nv w
          let s : Fin b := ⟨(l : ℕ) - (nv w - b), by have := l.2; omega⟩
          have hs : mark w s = l := Fin.ext (by rw [mark_val]; dsimp [s]; omega)
          constructor
          · simp only [graph, adj, extendAdj_cc]
            rw [← hr, ← hs]
            apply mark_clique w r s
            intro he
            have hil' := Fin.castSucc_lt_castSucc_iff.mp hil
            have hei : i = l := hr.symm.trans (he ▸ hs)
            exact ne_of_lt hil' hei
          · simp only [graph, adj, extendAdj_cl]
            exact ⟨s, hs⟩

end Word

/-- A marked NUIG with the paper's canonical ordered input and output cliques. -/
structure MarkedNUIG (a b : ℕ) where
  n : ℕ
  graph : SimpleGraph (Fin n)
  nuig : IsNUIG graph
  input_le : a ≤ n
  output_le : b ≤ n
  input_clique : ∀ i j : Fin n, (i : ℕ) < a → (j : ℕ) < a → i ≠ j → graph.Adj i j
  output_clique : ∀ i j : Fin n, n - b ≤ (i : ℕ) → n - b ≤ (j : ℕ) → i ≠ j → graph.Adj i j

namespace MarkedNUIG

variable {a b c : ℕ}

/-- A word realizes any marked NUIG; this choice affects no tableau matrix. -/
def word (G : MarkedNUIG a b) : Word a b := by
  classical
  exact (word_of_nuig G.nuig a b G.input_le G.output_le G.input_clique G.output_clique).choose

lemma word_spec (G : MarkedNUIG a b) : ∃ h : Word.nv G.word = G.n,
    (∀ u v, Word.adj G.word u v ↔ G.graph.Adj (Fin.cast h u) (Fin.cast h v)) ∧
    (∀ r, (Word.inp G.word r : ℕ) = r) ∧
    (∀ s, (Word.mark G.word s : ℕ) = G.n - b + s) := by
  classical
  exact (word_of_nuig G.nuig a b G.input_le G.output_le G.input_clique G.output_clique).choose_spec

/-- The matrix defined directly by tableaux on this graph. -/
def matrix (N : ℕ) (G : MarkedNUIG a b) : Matrix (Inj N b) (Inj N a) (KN N) :=
  hikMat N G.graph (Fin.castLE G.input_le) (fun s => ⟨G.n - b + s, by have := G.output_le; omega⟩)

lemma matrix_eq_word (N : ℕ) (G : MarkedNUIG a b) : G.matrix N = Word.tabMat N G.word := by
  obtain ⟨h, hadj, hi, ho⟩ := G.word_spec
  rw [← Word.hikMat_eq_tabMat]
  ext o i
  unfold matrix hikMat
  rw [hikSum_cast h _ _ hadj _ (Fin.castLE G.input_le)
    (fun r => Fin.ext (by simp [hi])) _ (fun s => ⟨G.n - b + s, by have := G.output_le; omega⟩)
    (fun s => Fin.ext (by simp [ho]))]

/-- Regard the graph of a word as a marked NUIG. -/
def ofWord (w : Word a b) : MarkedNUIG a b where
  n := Word.nv w
  graph := Word.graph w
  nuig := Word.graph_isNUIG w
  input_le := Word.a_le_nv w
  output_le := Word.level_le_nv w
  input_clique := by
    intro i j hi hj hne
    have hei : Word.inp w ⟨i, hi⟩ = i := Fin.ext (Word.inp_val w _)
    have hej : Word.inp w ⟨j, hj⟩ = j := Fin.ext (Word.inp_val w _)
    -- Inputs retain their initial clique.
    have H : ∀ {a b} (w : Word a b) (r s : Fin a), r ≠ s →
        Word.adj w (Word.inp w r) (Word.inp w s) := by
      intro a b w
      induction w with
      | nil => exact fun _ _ h => h
      | M w ih => simpa only [Word.inp, Word.adj, extendAdj_cc] using ih
      | D w ih => exact ih
    rw [← hei, ← hej]
    exact H w _ _ (fun he => hne (Fin.ext (congrArg (fun z : Fin a => (z : ℕ)) he)))
  output_clique := by
    intro i j hi hj hne
    have hb := Word.level_le_nv w
    let r : Fin b := ⟨(i : ℕ) - (Word.nv w - b), by have := i.2; omega⟩
    let s : Fin b := ⟨(j : ℕ) - (Word.nv w - b), by have := j.2; omega⟩
    have hr : Word.mark w r = i := Fin.ext (by rw [Word.mark_val]; dsimp [r]; omega)
    have hs : Word.mark w s = j := Fin.ext (by rw [Word.mark_val]; dsimp [s]; omega)
    rw [← hr, ← hs]
    exact Word.mark_clique w r s (by intro he; rw [he, hs] at hr; exact hne hr.symm)

lemma matrix_ofWord (N : ℕ) (w : Word a b) : (ofWord w).matrix N = Word.tabMat N w := by
  rw [← Word.hikMat_eq_tabMat]
  have hi : Fin.castLE (Word.a_le_nv w) = Word.inp w :=
    funext fun r => Fin.ext (Word.inp_val w r).symm
  have ho : (fun s : Fin b => (⟨Word.nv w - b + s, by
      have := Word.level_le_nv w; have := s.2; omega⟩ : Fin (Word.nv w))) = Word.mark w :=
    funext fun r => Fin.ext (Word.mark_val w r).symm
  dsimp only [matrix, ofWord]
  rw [hi, ho]

/-- Product of arbitrary marked NUIGs. `Word.adj_comp`, `Word.comp_overlap` and
`Word.comp_cover` identify this with gluing along the common boundary clique. -/
def comp (G : MarkedNUIG b c) (H : MarkedNUIG a b) : MarkedNUIG a c :=
  ofWord (Word.comp H.word G.word)

/-- **Lemma `tableau-mult`** for two arbitrary marked NUIGs. -/
theorem tableau_mult (N : ℕ) (G : MarkedNUIG b c) (H : MarkedNUIG a b) :
    (G.comp H).matrix N = G.matrix N * H.matrix N := by
  rw [comp, matrix_ofWord, Word.tabMat_comp, matrix_eq_word, matrix_eq_word]

end MarkedNUIG

/-- Set colours `N, …, M-1` to zero. -/
def restrictVars (N M : ℕ) : Pol M →ₐ[ℚ] Pol N :=
  aeval fun i => if h : (i : ℕ) < N then X ⟨i, h⟩ else 0

@[simp] lemma restrictVars_X {N M : ℕ} (i : Fin M) :
    restrictVars N M (X i) = if h : (i : ℕ) < N then X ⟨i, h⟩ else 0 := aeval_X _ _

lemma restrictVars_esymm {N M : ℕ} (h : N ≤ M) (k : ℕ) :
    restrictVars N M (esymm (Fin M) ℚ k) = esymm (Fin N) ℚ k := by
  classical
  let e : Fin N ↪ Fin M := ⟨Fin.castLE h, Fin.castLE_injective h⟩
  let s : Finset (Fin M) := univ.map e
  have hs : ∀ i : Fin M, i ∈ s ↔ (i : ℕ) < N := by
    intro i
    simp only [s, mem_map, mem_univ, true_and]
    exact ⟨by rintro ⟨j, rfl⟩; exact j.2,
      fun hi => ⟨⟨i, hi⟩, Fin.ext rfl⟩⟩
  unfold esymm
  rw [map_sum]
  trans ∑ t ∈ s.powersetCard k, restrictVars N M (∏ i ∈ t, X i)
  · symm
    apply sum_subset (fun _ ht => powersetCard_mono (subset_univ _) ht) ?_
    intro t ht hts
    have hnot : ¬t ⊆ s := by
      intro hsub
      exact hts (mem_powersetCard.mpr ⟨hsub, (mem_powersetCard.mp ht).2⟩)
    obtain ⟨i, hit, his⟩ := Set.not_subset.mp hnot
    rw [map_prod]
    apply prod_eq_zero hit
    rw [restrictVars_X, dite_eq_right (fun hx => his ((hs i).mpr hx))]
  · dsimp only [s]
    rw [powersetCard_map, sum_map]
    apply sum_congr rfl
    intro t _
    change restrictVars N M (∏ i ∈ t.map e, X i) = _
    rw [map_prod, prod_map]
    apply prod_congr rfl
    intro i _
    rw [restrictVars_X, dite_eq_left (show (e i : ℕ) < N from i.2)]
    rfl

lemma restrictVars_chromSym {V : Type*} [Fintype V] [DecidableEq V]
    {N M : ℕ} (h : N ≤ M) (G : SimpleGraph V) :
    restrictVars N M (chromSym M G) = chromSym N G := by
  classical
  let e : (V → Fin N) ↪ (V → Fin M) :=
    ⟨fun c v => Fin.castLE h (c v), fun c d he => funext fun v => Fin.ext (congrArg (fun f => (f v : ℕ)) he)⟩
  unfold chromSym
  rw [map_sum]
  symm
  apply Fintype.sum_of_injective e e.injective
  · intro c hc
    have hbad : ∃ v, ¬(c v : ℕ) < N := by
      by_contra! hall
      exact hc ⟨fun v => ⟨c v, hall v⟩, rfl⟩
    obtain ⟨v, hv⟩ := hbad
    split_ifs
    · rw [map_prod]
      exact prod_eq_zero (mem_univ v) (by rw [restrictVars_X, dite_eq_right hv])
    · simp
  · intro c
    have hp : IsProper G (e c) ↔ IsProper G c := by
      apply forall₂_congr
      intro u v
      exact imp_congr_right fun _ => (Fin.castLE_injective h).ne_iff
    rw [hp]
    split_ifs
    · rw [map_prod]
      exact prod_congr rfl fun v _ => by
        change X (c v) = restrictVars N M (X (Fin.castLE h (c v)))
        rw [restrictVars_X, dite_eq_left (show (Fin.castLE h (c v) : ℕ) < N from (c v).2)]
        rfl
    · simp

/-- `X_G` is homogeneous of degree `|V(G)|`. -/
lemma chromSym_homogeneous {V : Type*} [Fintype V] [DecidableEq V]
    (N : ℕ) (G : SimpleGraph V) : (chromSym N G).IsHomogeneous (Fintype.card V) := by
  classical
  unfold chromSym
  apply IsHomogeneous.sum
  intro c _
  split_ifs
  · simpa using IsHomogeneous.prod (univ : Finset V) (fun v => (X (c v) : Pol N))
      (fun _ => 1) (fun v _ => isHomogeneous_X ℚ (c v))
  · exact isHomogeneous_zero (n := Fintype.card V) (σ := Fin N) ℚ

lemma esymm_homogeneous (N k : ℕ) : (esymm (Fin N) ℚ k).IsHomogeneous k := by
  classical
  unfold esymm
  apply IsHomogeneous.sum
  intro t ht
  convert IsHomogeneous.prod t (fun i => (X i : Pol N)) (fun _ => 1)
    (fun i _ => isHomogeneous_X ℚ i) using 1
  simp [(mem_powersetCard.mp ht).2]

/-- Evaluation in elementary symmetric functions preserves weighted degree. -/
lemma elementary_monomial_homogeneous {N M : ℕ} (d : Fin N →₀ ℕ) (q : ℚ) :
    (aeval (fun j : Fin N => esymm (Fin M) ℚ ((j : ℕ) + 1)) (monomial d q)).IsHomogeneous
      (Finsupp.weight (fun j : Fin N => (j : ℕ) + 1) d) := by
  rw [aeval_monomial]
  apply IsHomogeneous.C_mul
  unfold Finsupp.prod
  convert IsHomogeneous.prod d.support (fun j => esymm (Fin M) ℚ ((j : ℕ) + 1) ^ d j)
    (fun j => ((j : ℕ) + 1) * d j) (fun j _ => (esymm_homogeneous M _).pow _) using 1
  simp only [Finsupp.weight_apply, Finsupp.sum, smul_eq_mul, mul_comm]

lemma elementary_homogeneousComponent {N M n : ℕ} (P : Pol N) :
    aeval (fun j : Fin N => esymm (Fin M) ℚ ((j : ℕ) + 1))
        (weightedHomogeneousComponent (fun j : Fin N => (j : ℕ) + 1) n P) =
      homogeneousComponent n (aeval (fun j : Fin N => esymm (Fin M) ℚ ((j : ℕ) + 1)) P) := by
  classical
  induction P using MvPolynomial.induction_on' with
  | add P Q hP hQ => simp only [map_add, hP, hQ]
  | monomial d q =>
    have hw := isWeightedHomogeneous_monomial (fun j : Fin N => (j : ℕ) + 1) d q rfl
    have hh := elementary_monomial_homogeneous (M := M) d q
    rw [weightedHomogeneousComponent_of_mem hw, homogeneousComponent_of_mem hh]
    split_ifs <;> simp

/-- The elementary expansion of a homogeneous symmetric polynomial is weighted homogeneous. -/
lemma elementary_homogeneous {N n : ℕ} {P : Pol N} {f : Pol N}
    (hf : f.IsHomogeneous n)
    (he : aeval (fun j : Fin N => esymm (Fin N) ℚ ((j : ℕ) + 1)) P = f) :
    P.IsWeightedHomogeneous (fun j : Fin N => (j : ℕ) + 1) n := by
  have hp : weightedHomogeneousComponent (fun j : Fin N => (j : ℕ) + 1) n P = P := by
    apply aeval_esymm_injective N
    rw [elementary_homogeneousComponent, he, homogeneousComponent_eq_self hf]
  rw [← hp]
  exact weightedHomogeneousComponent_isWeightedHomogeneous _ _

/-- Only `E_1, …, E_n` occur in weighted degree `n`. -/
lemma elementary_vars_bound {M n : ℕ} {P : Pol M}
    (hp : P.IsWeightedHomogeneous (fun j : Fin M => (j : ℕ) + 1) n)
    {j : Fin M} (hj : j ∈ P.vars) : (j : ℕ) < n := by
  obtain ⟨d, hd, hjd⟩ := (mem_vars_iff_mem_support j).mp hj

/-- **Finite-to-infinite step.** One polynomial in `E_1, …, E_|V|` works in every alphabet. -/
theorem stable_epositive {V : Type*} [Fintype V] [DecidableEq V] (G : SimpleGraph V)
    (hpos : ∀ N, ∃ P : Pol N, (∀ d, 0 ≤ P.coeff d) ∧
      aeval (fun j : Fin N => esymm (Fin N) ℚ ((j : ℕ) + 1)) P = chromSym N G) :
    ∃ P : Pol (Fintype.card V), (∀ d, 0 ≤ P.coeff d) ∧ ∀ N,
      aeval (fun j : Fin (Fintype.card V) => esymm (Fin N) ℚ ((j : ℕ) + 1)) P = chromSym N G := by

  let n := Fintype.card V
  obtain ⟨P, hp, he⟩ := hpos n

  let L := max n N

  have hNL : N ≤ L := le_max_right _ _

  obtain ⟨Q', hQ'⟩ := exists_rename_eq_of_vars_subset_range Q (Fin.castLE hnL)
    (Fin.castLE_injective hnL) (by
      intro j hj
      exact ⟨⟨j, elementary_vars_bound hhom hj⟩, Fin.ext rfl⟩)
  have hspec : ∀ K, K ≤ L →
      aeval (fun j : Fin n => esymm (Fin K) ℚ ((j : ℕ) + 1)) Q' = chromSym K G := by
    intro K hKL
    have ht := congrArg (restrictVars K L) hQ
    rw [← hQ', aeval_rename, restrictVars_chromSym hKL] at ht
    rw [← AlgHom.comp_apply, comp_aeval] at ht

  have hPQ : Q' = P := aeval_esymm_injective n ((hspec n hnL).trans he.symm)

  exact hspec N hNL

theorem specializeSymm_jointly_injective {P Q : Symm}
    (h : ∀ N, specializeSymm N P = specializeSymm N Q) : P = Q := by
  classical
  let n := (P - Q).vars.sup id + 1
  obtain ⟨R, hR⟩ := exists_rename_eq_of_vars_subset_range (P - Q)
    (fun j : Fin n => (j : ℕ)) Fin.val_injective (by
      intro j hj

  rw [← hR, specializeSymm, aeval_rename] at hz
  have hzero : R = 0 := aeval_esymm_injective n (by
    simpa only [Function.comp_def, map_zero] using hz)

  exact sub_eq_zero.mp hR.symm

/-!
# Equation `weight-pr` and Hikita's probability formula

The column potential is a shape invariant: a column of height `h` contributes
`((y+1) e_{y+1}/e_y)^h`. `row_potential` identifies this with the paper's product
of `λ_r! e_{λ_r}` over rows. This makes the telescoping argument independent of
how rows of equal length are ordered.
-/

noncomputable section

namespace ProperCircularArc

open Finset

local instance (N : ℕ) : CharZero (KN N) := Algebra.charZero_of_charZero ℚ (KN N)

/-- The probability of a column assignment under Hikita's rule at `q = 1`. -/
def hprob (N : ℕ) {n : ℕ} (G : SimpleGraph (Fin n)) (a : ℕ)
    (c : Fin n → Fin N) : ℚ :=
  ∏ v : Fin n, if a ≤ (v : ℕ) then
    Acoef ((c v : ℕ) + 1) (nbrCols G c v) / ((c v : ℕ) + 1) else 1

lemma hprob_nonneg {N n : ℕ} (G : SimpleGraph (Fin n)) (a : ℕ) (c : Fin n → Fin N) :
    0 ≤ hprob N G a c := by
  unfold hprob
  apply Finset.prod_nonneg
  intro v _
  split_ifs
  · exact div_nonneg (Acoef_nonneg _ _) (by positivity)
  · exact zero_le_one

/-- The factor contributed by a box in column `y+1`. -/
def boxFactor (N : ℕ) (y : Fin N) : KN N := ((y : ℕ) + 1) * Erat N y

lemma boxFactor_ne_zero {N : ℕ} (y : Fin N) : boxFactor N y ≠ 0 := by
  unfold boxFactor
  apply mul_ne_zero
  · exact_mod_cast (Nat.succ_ne_zero (y : ℕ))
  · exact Erat_ne_zero y.2

/-- Shape potential in terms of column heights (columns are zero based). -/
def shapePotential (N : ℕ) (h : ℕ → ℕ) : KN N :=
  ∏ y : Fin N, boxFactor N y ^ h y

lemma shapePotential_ne_zero {N : ℕ} (h : ℕ → ℕ) : shapePotential N h ≠ 0 :=
  Finset.prod_ne_zero_iff.mpr fun y _ => pow_ne_zero _ (boxFactor_ne_zero y)

lemma row_telescope {N : ℕ} (r : ℕ) (hr : r ≤ N) :
    (∏ y ∈ range r, (((y + 1 : ℕ) : KN N) * Erat N y)) =
      (r.factorial : KN N) * eK N r := by
  induction r with
  | zero => simp [eK]
  | succ r ih =>
    rw [prod_range_succ, ih (by omega), Nat.factorial_succ, Nat.cast_mul]
    have he : eK N r ≠ 0 := eK_ne_zero (by omega)
    unfold Erat
    field_simp

/-- For rows `rows`, the column potential is exactly `∏ r, r! e_r`. -/
theorem row_potential {N : ℕ} (rows : List ℕ) (hr : ∀ r ∈ rows, r ≤ N) :
    shapePotential N (fun y => (rows.filter (fun r => y < r)).length) =
      (rows.map (fun r => (r.factorial : KN N) * eK N r)).prod := by
  induction rows with
  | nil => simp [shapePotential]
  | cons r rows ih =>
    have hr' : r ≤ N := hr r (by simp)
    have hrows : ∀ s ∈ rows, s ≤ N := fun s hs => hr s (by simp [hs])
    have hcol : shapePotential N (fun y => ((r :: rows).filter (fun s => y < s)).length) =
        (∏ y : Fin N, if (y : ℕ) < r then boxFactor N y else 1) *
        shapePotential N (fun y => (rows.filter (fun s => y < s)).length) := by
      unfold shapePotential
      rw [← prod_mul_distrib]
      refine prod_congr rfl fun y _ => ?_
      by_cases hy : (y : ℕ) < r <;> simp [hy, pow_succ, mul_comm]
    rw [hcol, ih hrows, List.map_cons, List.prod_cons]
    congr 1
    have hp : (∏ y : Fin N, if (y : ℕ) < r then boxFactor N y else 1) =
        ∏ y ∈ range r, (((y + 1 : ℕ) : KN N) * Erat N y) := by
      rw [← Finset.prod_filter]
      exact Finset.prod_bij (fun y _ => (y : ℕ))
        (by intro y hy; simpa using (mem_filter.mp hy).2)
        (by intro _ _ _ _ h; exact Fin.ext h)
        (by intro y hy; exact ⟨⟨y, lt_of_lt_of_le (mem_range.mp hy) hr'⟩,
          by simpa using hy, rfl⟩) (by intro y _; simp [boxFactor])
    rw [hp, row_telescope r hr']

lemma added_potential {N n a : ℕ} (c : Fin n → Fin N) :
    shapePotential N (colH a c n) = shapePotential N (initH a c) *
      ∏ v : Fin n, if a ≤ (v : ℕ) then boxFactor N (c v) else 1 := by
  classical
  unfold shapePotential colH
  simp_rw [pow_add]
  rw [prod_mul_distrib]
  congr 1
  have hc : ∀ y : Fin N, boxFactor N y ^ addH a c n y =
      ∏ v : Fin n, if a ≤ (v : ℕ) ∧ c v = y then boxFactor N y else 1 := by
    intro y
    unfold addH
    rw [← prod_const, prod_filter]
    apply prod_congr rfl
    intro v _
    simp only [Fin.is_lt, true_and, Fin.ext_iff]
  simp_rw [hc]
  rw [prod_comm]
  refine prod_congr rfl fun v _ => ?_
  by_cases hv : a ≤ (v : ℕ)
  · simp [hv, eq_comm]
  · simp [hv]

/-- **Equation `weight-pr`**, including an arbitrary initial tableau `T_j`. -/
theorem weight_pr {N n : ℕ} (G : SimpleGraph (Fin n)) (a : ℕ) (c : Fin n → Fin N) :
    hwt N G a c = (hprob N G a c : KN N) *
      (shapePotential N (colH a c n) / shapePotential N (initH a c)) := by
  classical
  rw [added_potential]
  have hS := shapePotential_ne_zero (N := N) (initH a c)
  rw [mul_div_cancel_left₀ _ hS]
  unfold hwt hprob
  rw [Rat.cast_prod]
  simp only [apply_ite (fun q : ℚ => (q : KN N)), Rat.cast_one]
  rw [← prod_mul_distrib]
  refine prod_congr rfl fun v _ => ?_
  by_cases hv : a ≤ (v : ℕ)
  · simp only [hv, ite_true, Rat.cast_div, boxFactor]
    have hy : (((c v : ℕ) + 1 : ℕ) : KN N) ≠ 0 := Nat.cast_ne_zero.mpr (by omega)
    push_cast
    field_simp
  · simp [hv]

/-- Equation `weight-pr` in the paper's row notation, for the initial and final shapes. -/
theorem weight_pr_rows {N n : ℕ} (G : SimpleGraph (Fin n)) (a : ℕ) (c : Fin n → Fin N)
    (initial final : List ℕ) (hi : ∀ r ∈ initial, r ≤ N) (hf : ∀ r ∈ final, r ≤ N)
    (hci : ∀ y, initH a c y = (initial.filter (fun r => y < r)).length)
    (hcf : ∀ y, colH a c n y = (final.filter (fun r => y < r)).length) :
    hwt N G a c = (hprob N G a c : KN N) *
      ((final.map (fun r => (r.factorial : KN N) * eK N r)).prod /
       (initial.map (fun r => (r.factorial : KN N) * eK N r)).prod) := by
  rw [weight_pr, show colH a c n = _ from funext hcf,
    show initH a c = _ from funext hci, row_potential final hf, row_potential initial hi]

/-- Hikita's theorem in probability form; `row_potential` gives the row notation of the paper. -/
theorem hikita_probability {n : ℕ} (G : SimpleGraph (Fin n)) [DecidableRel G.Adj]
    (hG : IsNUIG G) (N : ℕ) :
    algebraMap (Pol N) (KN N) (chromSym N G) =
      ∑ c : Fin n → Fin N, (hprob N G 0 c : KN N) * shapePotential N (colH 0 c n) := by
  rw [hikita_weight G hG N]
  refine sum_congr rfl fun c _ => ?_
  rw [weight_pr]
  have hi : initH 0 c = fun _ => 0 := by funext y; simp [initH]
  simp [hi, shapePotential]

records = []


def weights(w):
    return '![' + ', '.join(map(str, w)) + ']'


def edges(E):
    return '{' + ', '.join(f'({a}, {b})' for a, b in E) + '}' if E else '∅'


def expression(q):
    if q == 0:
        return '0'
    terms = []
    for powers, coeff in sp.Poly(q, *P).terms():
        factors = []
        c = abs(int(coeff))
        if c != 1 or not any(powers):
            factors.append(str(c))
        for k, m in enumerate(powers):
            if m:
                factors.append(f'p N {k}' + (f' ^ {m}' if m != 1 else ''))
        term = ' * '.join(factors)
        terms.append(('-' if coeff < 0 else '+', term))
    first, *rest = terms
    return ('-' if first[0] == '-' else '') + first[1] + ''.join(f' {sign} {term}' for sign, term in rest)

def certificate(w, E):
    n = len(w)
    loop = next((a for a, b in E if a == b), None)
    if loop is not None:
        result, kind, data = sp.Integer(0), 'loop', loop
    elif not n:
        result, kind, data = sp.Integer(1), 'empty', None
    else:
        inc = [a for a, b in E if b == n - 1]
        if not inc:
            child, q = certificate(w[:-1], E)
            result, kind, data = sp.expand(P[w[-1]] * q), 'isolated', child
        else:
            u = inc[0]
            tail = tuple(x for x in E if x != (u, n - 1))
            neww = list(w[:-1])
            neww[u] += w[-1]
            def f(x):
                return u if x == n - 1 else x

            right, qr = certificate(tuple(neww), newe)
            result, kind, data = sp.expand(ql - qr), 'edge', (u, tail, tuple(neww), newe, left, right)
    i = len(records)
    records.append((w, E, result, kind, data))
    return i, result
